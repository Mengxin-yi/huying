## 二、状态管理优化

### 2.1 引入 Pinia

**建议：** 使用 Pinia 替代 localStorage 管理全局状态

```javascript
// store/modules/user.js
import { defineStore } from 'pinia';
import { userApi } from '@/api';

export const useUserStore = defineStore('user', {
  state: () => ({
    token: '',
    userInfo: null,
    permissions: [],
    vipInfo: null
  }),

  getters: {
    isLoggedIn: (state) => !!state.token,
    isVip: (state) => state.vipInfo?.is_valid || false
  },

  actions: {
    async login(credentials) {
      const res = await userApi.login(credentials);
      this.token = res.token;
      this.userInfo = res.user_info;
      this.permissions = res.permissions;
      uni.setStorageSync('token', res.token);
    },

    async logout() {
      await userApi.logout();
      this.$reset();
      uni.removeStorageSync('token');
    },

    async getProfile() {
      const res = await userApi.profile();
      this.userInfo = res;
    }
  }
});
```

### 3.2 组合式函数

```javascript
// composables/useList.js
import { ref, reactive } from 'vue';

export function useList(fetchFn, options = {}) {
  const list = ref([]);
  const loading = ref(false);
  const finished = ref(false);
  const pagination = reactive({
    page: 1,
    size: options.size || 20,
    total: 0
  });

  const load = async (refresh = false) => {
    if (refresh) {
      pagination.page = 1;
      finished.value = false;
    }

    if (finished.value) return;

    loading.value = true;
    try {
      const res = await fetchFn({
        page: pagination.page,
        size: pagination.size,
        ...options.params
      });

      if (refresh) {
        list.value = res.items;
      } else {
        list.value.push(...res.items);
      }

      pagination.total = res.total;
      finished.value = list.value.length >= res.total;
      pagination.page++;
    } finally {
      loading.value = false;
    }
  };

  const refresh = () => load(true);

  return { list, loading, finished, pagination, load, refresh };
}
```

```javascript
// composables/useUpload.js
export function useUpload(options = {}) {
  const uploading = ref(false);
  const progress = ref(0);
  const fileList = ref([]);

  const chooseImage = (count = 9) => {
    return new Promise((resolve, reject) => {
      uni.chooseImage({
        count,
        sizeType: ['compressed'],
        sourceType: ['album', 'camera'],
        success: (res) => resolve(res.tempFilePaths),
        fail: reject
      });
    });
  };

  const upload = async (filePath) => {
    uploading.value = true;
    try {
      const res = await uploadApi.upload(filePath, (p) => {
        progress.value = p;
      });
      fileList.value.push(res.url);
      return res.url;
    } finally {
      uploading.value = false;
      progress.value = 0;
    }
  };

  const remove = (index) => {
    fileList.value.splice(index, 1);
  };

  return { uploading, progress, fileList, chooseImage, upload, remove };
}
```

### 5.3 请求优化

```javascript
// 请求缓存
const cache = new Map();

export function cachedRequest(key, fetchFn, ttl = 5 * 60 * 1000) {
  if (cache.has(key)) {
    const { data, expire } = cache.get(key);
    if (Date.now() < expire) {
      return Promise.resolve(data);
    }
  }

  return fetchFn().then((data) => {
    cache.set(key, {
      data,
      expire: Date.now() + ttl
    });
    return data;
  });
}

// 请求去重
const pendingRequests = new Map();

export function dedupeRequest(key, fetchFn) {
  if (pendingRequests.has(key)) {
    return pendingRequests.get(key);
  }

  const promise = fetchFn().finally(() => {
    pendingRequests.delete(key);
  });

  pendingRequests.set(key, promise);
  return promise;
}
```

## 九、重构优先级

### 高优先级

1. **API 模块化** - 拆分 webapi.js，按业务模块组织
2. **状态管理** - 引入 Pinia，统一状态管理
3. **类型安全** - 添加 TypeScript 支持

### 中优先级

4. **组件提取** - 提取通用组件，减少重复代码
5. **性能优化** - 虚拟列表、图片压缩
6. **安全优化** - 移除硬编码敏感信息

### 低优先级

7. **测试覆盖** - 添加单元测试、E2E 测试
8. **开发规范** - 统一代码风格、提交规范
